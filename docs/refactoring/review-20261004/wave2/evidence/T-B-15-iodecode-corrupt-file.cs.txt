// Evidence for T-B-15 (deterministic sketch, never compiled). Needs the WPF decoder, so it belongs in the UI/Integration tests.
// IoDecodeSplit.RunAsync (tools/PhotoReview.Benchmark.Cli/IoDecodeSplit.cs:68-76) awaits MeasureFile per file without a catch;
// MeasureFile calls BitmapDecoder.Create / DecodeFromMemory which throw NotSupportedException or FileFormatException
// (not IOException) for a truncated or corrupt image. Program.cs:164-169 only catches DirectoryNotFoundException and
// InvalidOperationException, so the exception is unhandled, raw.csv/summary.md are never written (they are produced after the loop).
[Fact]
public async Task CorruptImage_AbortsWholeRun_WithoutReports()
{
    using var dir = new TempDirectory();
    File.WriteAllBytes(dir.Combine("a.jpg"), ValidTinyJpeg());
    File.WriteAllBytes(dir.Combine("b.jpg"), new byte[] { 0xFF, 0xD8, 0xFF, 0x00 }); // truncated JPEG
    var outDir = dir.Combine("out");
    await Assert.ThrowsAnyAsync<Exception>(() => IoDecodeSplit.RunAsync(dir.Path, outDir, [0], 10));
    Assert.False(File.Exists(Path.Combine(outDir, "summary.md")));   // good file a.jpg is lost with the bad one
}
