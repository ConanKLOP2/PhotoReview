using System.Text.Json;
using PhotoReview.Localization.Generator;

var o = new JsonDocumentOptions { CommentHandling = JsonCommentHandling.Skip, AllowTrailingCommas = true };
for (var n = 60; n <= 68; n++)
{
    var text = "{\"_m\":" + new string('[', n) + new string(']', n) + ",\"a\":\"x\"}";
    bool stj; try { using var d = JsonDocument.Parse(text, o); stj = true; } catch (JsonException) { stj = false; }
    bool gen; try { FlatJsonReader.Read(text); gen = true; } catch (FlatJsonException) { gen = false; }
    Console.WriteLine($"array depth {n} (+root = {n + 1}): STJ={(stj ? "accept" : "reject")} generator={(gen ? "accept" : "reject")}");
}
