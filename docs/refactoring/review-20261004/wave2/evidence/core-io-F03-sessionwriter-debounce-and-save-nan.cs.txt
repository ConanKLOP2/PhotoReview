using PhotoReview.Core;
using PhotoReview.Core.IO;
using PhotoReview.Core.Session;
using PhotoReview.Core.Settings;
using PhotoReview.Core.Diagnostics;
var t = Path.Combine(Path.GetTempPath(), "prrev2_" + Guid.NewGuid().ToString("N"));
var paths = new AppPaths(t, t, true);
var fs = new PhysicalFileSystem();
// 1) Save(NaN)
var ss = new SettingsStore(paths, fs, NullLog.Instance);
try { ss.Save(new AppSettings { InfoOverlayFontSize = double.NaN }); Console.WriteLine("Save(NaN): no throw"); }
catch (Exception ex) { Console.WriteLine("Save(NaN) threw " + ex.GetType().Name + ": " + ex.Message); }
Console.WriteLine("config exists after NaN save: " + File.Exists(paths.ConfigFile));
// 2) SessionWriter negative debounce
var store = new SessionStore(paths, fs);
var w = new SessionWriter(store, null, TimeSpan.FromMilliseconds(-5));
w.Update(new SessionState { Folder = t, CurrentPath = "a.jpg" });
await Task.Delay(500);
Console.WriteLine("session written after 500ms with debounce=-5ms: " + File.Exists(store.GetPath(t)));
Console.WriteLine("WhenIdleAsync faulted: " + w.WhenIdleAsync().IsFaulted);
w.Update(new SessionState { Folder = t, CurrentPath = "b.jpg" });
await Task.Delay(300);
Console.WriteLine("still not written after 2nd update: " + !File.Exists(store.GetPath(t)));
w.Flush();
Console.WriteLine("written after explicit Flush: " + File.Exists(store.GetPath(t)));
w.Dispose();
// 3) zero debounce => write happens inside Update on the caller thread
var w2 = new SessionWriter(store, null, TimeSpan.Zero);
var before = File.Exists(store.GetPath(Path.Combine(t,"z")));
w2.Update(new SessionState { Folder = Path.Combine(t,"z"), CurrentPath = "z.jpg" });
Console.WriteLine("debounce=0: file already exists when Update returns: " + File.Exists(store.GetPath(Path.Combine(t,"z"))) + " (before=" + before + ")");
w2.Dispose();
try { Directory.Delete(t, true); } catch {}
